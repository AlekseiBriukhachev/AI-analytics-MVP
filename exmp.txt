package com.aianalytics.ml

import kotlin.math.exp
import kotlin.math.sqrt
import kotlin.random.Random

class AdvancedNeuralNetwork(
    private val layers: List<Int>,
    private val learningRate: Double = 0.01,
    private val activationFunction: String = "sigmoid"
) {
    private val weights: List<Array<DoubleArray>>
    private val biases: List<DoubleArray>
    private val random = Random.Default

    init {
        weights = mutableListOf<Array<DoubleArray>>().apply {
            for (i in 0 until layers.size - 1) {
                val layerWeights = Array(layers[i]) { DoubleArray(layers[i + 1]) }
                val xavier = sqrt(2.0 / layers[i])
                for (j in 0 until layers[i]) {
                    for (k in 0 until layers[i + 1]) {
                        layerWeights[j][k] = random.nextGaussian() * xavier
                    }
                }
                add(layerWeights)
            }
        }

        biases = layers.drop(1).map { size ->
            DoubleArray(size) { random.nextGaussian() * 0.1 }
        }
    }

    private fun activate(x: Double): Double = when (activationFunction) {
        "sigmoid" -> 1.0 / (1.0 + exp(-x))
        "relu" -> maxOf(0.0, x)
        "tanh" -> kotlin.math.tanh(x)
        else -> 1.0 / (1.0 + exp(-x))
    }

    private fun activateDerivative(x: Double): Double = when (activationFunction) {
        "sigmoid" -> x * (1 - x)
        "relu" -> if (x > 0) 1.0 else 0.0
        "tanh" -> 1.0 - x * x
        else -> x * (1 - x)
    }

    fun feedForward(input: DoubleArray): List<DoubleArray> {
        val activations = mutableListOf<DoubleArray>()
        var currentInput = input
        activations.add(currentInput)

        for (layerIndex in weights.indices) {
            val layerOutput = DoubleArray(layers[layerIndex + 1])
            for (j in layerOutput.indices) {
                layerOutput[j] = biases[layerIndex][j]
                for (i in currentInput.indices) {
                    layerOutput[j] += currentInput[i] * weights[layerIndex][i][j]
                }
                layerOutput[j] = activate(layerOutput[j])
            }
            currentInput = layerOutput
            activations.add(currentInput)
        }

        return activations
    }

    fun predict(input: DoubleArray): Double {
        val activations = feedForward(input)
        return activations.last()[0]
    }

    fun train(inputs: List<DoubleArray>, targets: List<DoubleArray>, epochs: Int): TrainingStats {
        var totalLoss = 0.0
        val startTime = System.currentTimeMillis()

        repeat(epochs) { epoch ->
            var epochLoss = 0.0

            for (i in inputs.indices) {
                val activations = feedForward(inputs[i])
                val output = activations.last()

                // Calculate loss
                val loss = targets[i].mapIndexed { j, target ->
                    val error = target - output[j]
                    epochLoss += error * error
                    error
                }.toDoubleArray()

                // Backpropagation
                backpropagate(activations, loss)
            }

            if (epoch % 100 == 0) {
                totalLoss = epochLoss / inputs.size
                println("Epoch $epoch, Loss: $totalLoss")
            }
        }

        val endTime = System.currentTimeMillis()
        return TrainingStats(
            finalLoss = totalLoss,
            trainingTime = endTime - startTime
        )
    }

    private fun backpropagate(activations: List<DoubleArray>, outputError: DoubleArray) {
        val errors = mutableListOf<DoubleArray>()
        errors.add(0, outputError.map { it * activateDerivative(activations.last()[0]) }.toDoubleArray())

        // Calculate errors for hidden layers
        for (layerIndex in weights.size - 2 downTo 0) {
            val layerError = DoubleArray(layers[layerIndex + 1])
            for (j in layerError.indices) {
                layerError[j] = 0.0
                for (k in errors[0].indices) {
                    layerError[j] += errors[0][k] * weights[layerIndex + 1][j][k]
                }
                layerError[j] *= activateDerivative(activations[layerIndex + 1][j])
            }
            errors.add(0, layerError)
        }

        // Update weights and biases
        for (layerIndex in weights.indices) {
            for (i in weights[layerIndex].indices) {
                for (j in weights[layerIndex][i].indices) {
                    weights[layerIndex][i][j] += learningRate * errors[layerIndex][j] * activations[layerIndex][i]
                }
            }

            for (j in biases[layerIndex].indices) {
                biases[layerIndex][j] += learningRate * errors[layerIndex][j]
            }
        }
    }
}

data class TrainingStats(
    val finalLoss: Double,
    val trainingTime: Long
)

// src/main/kotlin/com/aianalytics/services/MLService.kt
package com.aianalytics.services

import com.aianalytics.ml.AdvancedNeuralNetwork
import com.aianalytics.models.*
import java.util.*
import java.util.concurrent.ConcurrentHashMap

class MLService {
    private val datasets = ConcurrentHashMap<String, Dataset>()
    private val models = ConcurrentHashMap<String, AdvancedNeuralNetwork>()
    private val modelConfigs = ConcurrentHashMap<String, ModelConfig>()

    fun uploadDataset(dataset: Dataset): String {
        val id = UUID.randomUUID().toString()
        datasets[id] = dataset
        return id
    }

    fun getDataset(id: String): Dataset? = datasets[id]

    fun getAllDatasets(): List<Pair<String, Dataset>> = datasets.toList()

    suspend fun trainModel(request: TrainingRequest): TrainingResult {
        val dataset = datasets[request.datasetId]
            ?: throw IllegalArgumentException("Dataset not found")

        if (dataset.data.any { it.label == null }) {
            throw IllegalArgumentException("Dataset must have labels for supervised learning")
        }

        val modelId = UUID.randomUUID().toString()
        val inputSize = dataset.data.first().features.size
        val layers = listOf(inputSize) + request.config.hiddenLayers + listOf(1)

        val model = AdvancedNeuralNetwork(
            layers = layers,
            learningRate = request.config.learningRate,
            activationFunction = "sigmoid"
        )

        // Prepare training data
        val shuffledData = dataset.data.shuffled()
        val splitIndex = (shuffledData.size * (1 - request.testSplit)).toInt()
        val trainData = shuffledData.take(splitIndex)
        val testData = shuffledData.drop(splitIndex)

        val trainInputs = trainData.map { it.features.toDoubleArray() }
        val trainTargets = trainData.map { doubleArrayOf(it.label!!) }

        // Train the model
        val stats = model.train(trainInputs, trainTargets, request.config.epochs)

        // Calculate accuracy on test set
        var correct = 0
        testData.forEach { testPoint ->
            val prediction = model.predict(testPoint.features.toDoubleArray())
            val predicted = if (prediction > 0.5) 1.0 else 0.0
            if (predicted == testPoint.label) correct++
        }

        val accuracy = if (testData.isNotEmpty()) correct.toDouble() / testData.size else 0.0

        models[modelId] = model
        modelConfigs[modelId] = request.config

        return TrainingResult(
            modelId = modelId,
            accuracy = accuracy,
            loss = stats.finalLoss,
            epochs = request.config.epochs,
            trainingTime = stats.trainingTime
        )
    }

    fun predict(request: PredictionRequest): PredictionResult {
        val model = models[request.modelId]
            ?: throw IllegalArgumentException("Model not found")

        val prediction = model.predict(request.features.toDoubleArray())
        val confidence = kotlin.math.abs(prediction - 0.5) * 2 // Simple confidence metric

        return PredictionResult(prediction, confidence)
    }

    fun getModelsList(): List<String> = models.keys.toList()
}

package com.aianalytics

import com.aianalytics.models.*

import io.ktor.server.plugins.statciccontent.*
import io.ktor.server.request.*
import io.ktor.server.response.*
import io.ktor.server.routing.*
import kotlinx.serialization.json.Json

fun main() {
    embeddedServer(Netty, port = 8080, host = "0.0.0.0", module = Application::module)
        .start(wait = true)
}

fun Application.module() {
    val mlService = MLService()

    install(ContentNegotiation) {
        json(Json {
            prettyPrint = true
            isLenient = true
            ignoreUnknownKeys = true
        })
    }

    install(CORS) {
        anyHost()
        allowHeader(HttpHeaders.ContentType)
        allowMethod(HttpMethod.Options)
        allowMethod(HttpMethod.Put)
        allowMethod(HttpMethod.Delete)
        allowMethod(HttpMethod.Patch)
    }

    // Serve static files
    staticResources("/", "static")

    routing {
        // API Routes
        route("/api") {

            // Upload dataset
            post("/datasets") {
                try {
                    val dataset = call.receive<Dataset>()
                    val id = mlService.uploadDataset(dataset)
                    call.respond(ApiResponse(true, mapOf("id" to id)))
                } catch (e: Exception) {
                    call.respond(
                        HttpStatusCode.BadRequest,
                        ApiResponse<Nothing>(false, error = e.message)
                    )
                }
            }

            // Get all datasets
            get("/datasets") {
                val datasets = mlService.getAllDatasets()
                    .map { (id, dataset) ->
                        mapOf(
                            "id" to id,
                            "name" to dataset.name,
                            "description" to dataset.description,
                            "size" to dataset.data.size,
                            "features" to dataset.data.firstOrNull()?.features?.size
                        )
                    }
                call.respond(ApiResponse(true, datasets))
            }

            // Train model
            post("/models/train") {
                try {
                    val request = call.receive<TrainingRequest>()
                    val result = mlService.trainModel(request)
                    call.respond(ApiResponse(true, result))
                } catch (e: Exception) {
                    call.respond(
                        HttpStatusCode.BadRequest,
                        ApiResponse<Nothing>(false, error = e.message)
                    )
                }
            }

            // Make prediction
            post("/models/predict") {
                try {
                    val request = call.receive<PredictionRequest>()
                    val result = mlService.predict(request)
                    call.respond(ApiResponse(true, result))
                } catch (e: Exception) {
                    call.respond(
                        HttpStatusCode.BadRequest,
                        ApiResponse<Nothing>(false, error = e.message)
                    )
                }
            }

            // Get models list
            get("/models") {
                val models = mlService.getModelsList()
                call.respond(ApiResponse(true, models))
            }

            // Health check
            get("/health") {
                call.respond(ApiResponse(true, mapOf("status" to "healthy")))
            }
        }

        // Serve index.html for root
        get("/") {
            call.respondText(indexHtml, ContentType.Text.Html)
        }
    }
}

                        config: { hiddenLayers, learningRate, epochs },
                        testSplit
                    })
                });

                const result = await response.json();
                if (result.success) {
                    const data = result.data;
                    showResult('trainingResult', `
                        <h3>✅ Модель обучена успешно!</h3>
                        <p><strong>ID модели:</strong> ${data.modelId}</p>

                        <p><strong>Эпох:</strong> ${data.epochs}</p>
                        <p><strong>Время обучения:</strong> ${data.trainingTime}мс</p>
                    `, 'success');

        async function makePrediction() {

                showResult('predictionResult', 'Заполните все поля', 'error');

#!/bin/bash

BASE_URL="http://localhost:8080/api"

echo "🧪 Тестирование API..."

# 1. Health Check
echo "1. Health Check..."
curl -s "$BASE_URL/health" | jq .

# 2. Загрузка тестового датасета
echo "2. Загрузка датасета..."
DATASET_RESPONSE=$(curl -s -X POST "$BASE_URL/datasets" \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Test Dataset",
    "description": "Тестовый датасет для проверки API",
    "data": [
      {"features": [0, 0], "label": 0},
      {"features": [0, 1], "label": 0},
      {"features": [1, 0], "label": 0},
      {"features": [1, 1], "label": 1}
    ]
  }')

echo $DATASET_RESPONSE | jq .
DATASET_ID=$(echo $DATASET_RESPONSE | jq -r '.data.id')

# 3. Обучение модели
echo "3. Обучение модели..."
TRAINING_RESPONSE=$(curl -s -X POST "$BASE_URL/models/train" \
  -H "Content-Type: application/json" \
  -d "{
    \"datasetId\": \"$DATASET_ID\",
    \"config\": {
      \"hiddenLayers\": [4],
      \"learningRate\": 0.5,
      \"epochs\": 1000
    },
    \"testSplit\": 0.25
  }")

echo $TRAINING_RESPONSE | jq .
MODEL_ID=$(echo $TRAINING_RESPONSE | jq -r '.data.modelId')

# 4. Предсказание
echo "4. Предсказание..."
curl -s -X POST "$BASE_URL/models/predict" \
  -H "Content-Type: application/json" \
  -d "{
    \"modelId\": \"$MODEL_ID\",
    \"features\": [1, 1]
  }" | jq .

echo "✅ Тестирование завершено!"
